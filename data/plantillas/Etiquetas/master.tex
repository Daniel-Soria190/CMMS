% ---- Comando de etiqueta ----
\newcommand{\etiqueta}[5]{
	\noindent
	\begin{tcolorbox}[
		width=\linewidth,
		height=6cm,
		colframe=black,
		colback=white,
		boxrule=0.8pt,
		sharp corners,
		valign=center,
		breakable=false
		]
		
		\begin{minipage}{0.6\linewidth}
			\begin{tcolorbox}[boxrule=0.5pt, left=2pt, right=2pt, top=2pt, bottom=2pt]
				\textbf{Marca:}\\ #1
			\end{tcolorbox}
			
			\begin{tcolorbox}[boxrule=0.5pt, left=2pt, right=2pt, top=2pt, bottom=2pt]
				\textbf{Modelo:}\\ #2
			\end{tcolorbox}
			
			\begin{tcolorbox}[boxrule=0.5pt, left=2pt, right=2pt, top=2pt, bottom=2pt]
				\textbf{Número de Serie:}\\ #3
			\end{tcolorbox}
			
			\begin{tcolorbox}[boxrule=0.5pt, left=2pt, right=2pt, top=2pt, bottom=2pt]
				\textbf{Área:}\\ #4
			\end{tcolorbox}
		\end{minipage}
		\hfill
		\begin{minipage}{0.35\linewidth}
			\centering
			\fancyqr{#5}
		\end{minipage}
		
	\end{tcolorbox}
	\vspace{2mm}
}
